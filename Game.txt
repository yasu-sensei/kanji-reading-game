  // 提供CSVの対象漢字が初めて登場する順。問題番号・CSV本体は変更しない。
  const kanji = Array.from("木大小一二三四五六七八九十子空先生男女手天青文字正見学校山水雨上下日火田川竹月車人気糸目玉村白土音花休虫金本森出中町入赤耳王口年立草名夕百円千犬早貝林右足石左力");
  const headers = ["番号", "対象漢字", "出題語彙", "正しい読み", "間違い①（候補）", "間違い②（候補）", "確認状況"];

  function parseCSV(text) {
    const rows = [];
    let row = [], field = "", quoted = false, closed = false;
    const source = text.replace(/^\uFEFF/, "");
    for (let index = 0; index < source.length; index += 1) {
      const char = source[index];
      if (quoted) {
        if (char === '"' && source[index + 1] === '"') { field += '"'; index += 1; }
        else if (char === '"') { quoted = false; closed = true; }
        else field += char;
      } else if (char === '"') {
        if (field || closed) throw new Error("不正な引用符です");
        quoted = true;
      } else if (char === ",") {
        row.push(field); field = ""; closed = false;
      } else if (char === "\n" || char === "\r") {
        if (char === "\r" && source[index + 1] === "\n") index += 1;
        row.push(field);
        if (row.some((value) => value !== "")) rows.push(row);
        row = []; field = ""; closed = false;
      } else {
        if (closed) throw new Error("引用符の後に不正な文字があります");
        field += char;
      }
    }
    if (quoted) throw new Error("引用符が閉じていません");
    if (field || closed || row.length) { row.push(field); rows.push(row); }
    return rows;
  }

  function parseQuestions(text) {
    const rows = parseCSV(text);
    if (JSON.stringify(rows.shift()) !== JSON.stringify(headers) || rows.length !== 400) throw new Error("400問の列構成と一致しません");
    const ids = new Set();
    const counts = new Map();
    const questions = rows.map((row) => {
      if (row.length !== headers.length || row.some((value) => !value.trim())) throw new Error("問題に空欄があります");
      const [number, target, word, correct, wrong1, wrong2, status] = row;
      const id = Number(number);
      if (!Number.isInteger(id) || id < 1 || id > 400 || ids.has(id) || !kanji.includes(target) || !word.includes(target) || new Set([correct, wrong1, wrong2]).size !== 3) throw new Error("問題番号・漢字・選択肢が不正です");
      ids.add(id);
      counts.set(target, (counts.get(target) || 0) + 1);
      return Object.freeze({ id, target, word, correct, choices: Object.freeze([correct, wrong1, wrong2]), status });
    });
    if (counts.size !== 80 || [...counts.values()].some((count) => count !== 5)) throw new Error("80字×5問と一致しません");
    return questions;
  }

  function shuffle(items) {
    const result = [...items];
    for (let index = result.length - 1; index > 0; index -= 1) {
      const random = new Uint32Array(1);
      crypto.getRandomValues(random);
      const other = Math.floor((random[0] / 4294967296) * (index + 1));
      [result[index], result[other]] = [result[other], result[index]];
    }
    return result;
  }

  window.KanjiQuestions = Object.freeze({ kanji: Object.freeze(kanji), parseCSV, parseQuestions, shuffle });

  function normalize(raw) {
    if (!raw || typeof raw !== "object" || Array.isArray(raw) || !raw.users || typeof raw.users !== "object" || Array.isArray(raw.users)) throw new Error("ほぞんデータを よみこめません。おとなの方に しらせてね。");

    }
    result.active ??= "";
    if (result.active && (!profilePattern.test(result.active) || !Object.hasOwn(result.users, result.active))) throw new Error("とうろくを たしかめてね。");
    return result;
  }
  function read() {
    const raw = localStorage.getItem(key);
    return raw === null ? { users: {}, active: "" } : normalize(JSON.parse(raw));
  }
  async function transact(change) {
    const run = () => {
      const next = read();
      const result = change(next);
      const checked = normalize(next);
      localStorage.setItem(key, JSON.stringify(checked));
      return { data: checked, result };
    };
    if (navigator.locks?.request) return navigator.locks.request(key, run);
    return run();
  }
  function profile(data, id) {
    if (!id || data.active !== id || !Object.hasOwn(data.users, id)) throw new Error("とうろくが かわったよ。じゅんびがめんに もどってね。");
    return data.users[id];
  }

  const supportingReadings = { 曜: "よう", 葉: "は", 声: "ごえ", 頭: "とう", 長: "ちょう", 紙: "し", 国: "ごく", 才: "さい", 色: "いろ", 作: "さく", 数: "すう", 門: "もん", 道: "みち", 地: "ち", 毎: "まい", 記: "き", 光: "こう", 電: "でん", 元: "げん", 毛: "け", 鳥: "ちょう", 夏: "なつ", 魚: "ぎょ", 心: "しん", 来: "らい", 原: "げん", 方: "がた", 朝: "ちょう", 遠: "えん", 体: "たい" };
  function questionWord(q) {
    return Array.from(q.w, char => {
      let reading = supportingReadings[char];
      if (q.id === 100 && char === "紙") reading = "がみ";
      if (q.id === 373 && char === "道") reading = "どう";
      return reading ? `<ruby>${esc(char)}<rp>（</rp><rt>${reading}</rt><rp>）</rp></ruby>` : esc(char);
    }).join("");
  }
  const shuffle = window.KanjiQuestions.shuffle;
  const today = () => new Intl.DateTimeFormat("sv-SE", { timeZone: "Asia/Tokyo", year: "numeric", month: "2-digit", day: "2-digit" }).format(new Date());
  const practiceToday = u => u.log.filter(log => log.mode === "practice" && log.date && new Intl.DateTimeFormat("sv-SE", { timeZone: "Asia/Tokyo", year: "numeric", month: "2-digit", day: "2-digit" }).format(new Date(log.date)) === today()).length;
  const user = () => data.users[data.active];
  const b = (text, action, classes = "") => `<button type="button" class="${classes}" onclick="${esc(action)}" ${saving ? "disabled" : ""}>${text}</button>`;

  async function update(change) {
    if (saving) return false;

    catch (error) { warning = "ほぞんできませんでした。星やカードは かえていません。 " + (error.name === "Error" ? error.message : "端末の ほぞんせっていを おとなの方に たしかめてもらってね。"); return false; }
    finally { saving = false; }
  }

    render();

    if (game && !game.committed && !confirm("とちゅうの もんだいは きろくされません。もどりますか？")) return;

  function available() { const set = new Set(user().selected); return QUESTIONS.filter(q => set.has(q.k)); }
  function poolQuestions(total) {
    const all = available(), history = user().history;
    // 添付試作版の暫定配分を維持。正式な習熟度の計算は未確定。
    const unseen = shuffle(all.filter(q => !history[q.id]));
    const weak = shuffle(all.filter(q => history[q.id]?.wrong > 0)).sort((a, b) => history[b.id].wrong - history[a.id].wrong);
    const rest = shuffle(all.filter(q => history[q.id]));
    const chosen = [], ids = new Set();
    function add(items, n) { for (const q of items) { if (chosen.length >= total || n <= 0) break; if (ids.has(q.id)) continue; chosen.push(q); ids.add(q.id); n--; } }
    add(unseen, 5); add(weak, 3); add(rest, total - chosen.length); add(shuffle(all), total - chosen.length);
    while (chosen.length < total) chosen.push(...shuffle(all).slice(0, total - chosen.length));
    return shuffle(chosen).map(q => ({ ...q, opts: shuffle([q.r, ...q.wrong]) }));
  }
  function begin(mode) {

  }
  async function commitGame() {
    if (game.committed) return true;
    const current = game;
    const score = current.answers.filter(x => x.ok).length;
    let award = 0;
    const ok = await update(next => {
      const u = store.profile(next, current.profile);
      const prior = u.log.find(log => log.id === current.id);
      if (prior) { award = prior.award; return; }
      if (current.mode === "homework") {
        if (u.lastHomework === today()) throw new Error("きょうの しゅくだいは すでに きろくされています。");
        award = score === current.items.length ? 2 : score >= current.items.length / 2 ? 1 : 0; u.lastHomework = today();
      } else { if (practiceToday(u) >= 5) throw new Error("きょうの れんしゅうは 5かい おわっています。また あした！"); u.practice++; if (u.practice >= 5) { award = 1; u.practice = 0; } }
      for (const answer of current.answers) { const h = u.history[answer.id] ?? { seen: 0, correct: 0, wrong: 0 }; h.seen++; if (answer.ok) h.correct++; else h.wrong++; u.history[answer.id] = h; }
      u.stars += award;
      u.log.push({ id: current.id, date: new Date().toISOString(), mode: current.mode, score, award });
    });
    if (ok) { current.committed = true; current.awarded = award; }
    return ok;
  }
  async function answer(index) {

    if (current.answers.length === current.items.length) { await commitGame(); if (game !== current) return; render(); }

  }
  async function retryCommit() { if (await commitGame()) warning = ""; render(); }
  function quit() { if (game?.committed) { nav("result"); return; } if (!confirm("とちゅうで やめる？ さいごまで おわっていないと きろくは のこらないよ。")) return; cancelTimer(); game = review = feedback = null; screen = "select"; render(); }
  function startReview() {
    if (!game?.committed) return;
    const missed = game.items.filter((q, i) => !game.answers[i].ok);
    const unique = [...new Map(missed.map(q => [q.id, q])).values()];
    if (!unique.length) return;
    review = { round: 1, items: shuffle(unique.map(q => ({ ...q, opts: shuffle([q.r, ...q.wrong]) }))), idx: 0, failed: [] };
    feedback = null; screen = "review"; render();
  }
  function reviewAnswer(index) {
    if (feedback || screen !== "review") return;
    const q = review.items[review.idx], ok = q.opts[index] === q.r;
    feedback = { ok, choice: index }; if (!ok) review.failed.push(q); render(); cancelTimer();
    timer = setTimeout(() => {
      if (screen !== "review") return;
      feedback = null; review.idx++;
      if (review.idx >= review.items.length) {
        if (review.round === 1 && review.failed.length) review = { round: 2, items: shuffle(review.failed.map(q => ({ ...q, opts: shuffle([q.r, ...q.wrong]) }))), idx: 0, failed: [] };
        else { game.reviewFinished = true; screen = "reviewDone"; }
      }
      render();
    }, ok ? 1500 : 2000);
  }
  function nav(to) { cancelTimer(); feedback = review = null; if (to !== "result") game = null; screen = to; warning = ""; render(); }

    }

    html += `<div class="row user-header"><span class="badge">${esc(data.active.replace("-", "くみ "))}ばん</span><span class="badge" id="learning-stars">⭐ ${u.stars}こ</span>${b("とうろく きりかえ", "changeUser()", "small secondary")}</div>`;
    if (screen === "home") screen = "select";
    if (screen === "select") html += `<section class="card"><h2>① ならった かんじを えらぼう！</h2><p class="muted">タッチで えらぶ・もういちどで はずす</p><div class="row"><span class="badge">${u.selected.length} / 80じ</span><div class="buttons">${b("ぜんぶ えらぶ", "selectAll(true)", "small secondary")}${b("ぜんぶ はずす", "selectAll(false)", "small secondary")}</div></div><div class="grid">${KANJI.map(k => `<button type="button" class="kanji ${u.selected.includes(k) ? "on" : ""}" aria-label="${k}" aria-pressed="${u.selected.includes(k)}" onclick="toggle('${k}')" ${saving ? "disabled" : ""}>${k}</button>`).join("")}</div><p class="muted">えらんだ かんじの ことば：${available().length}もん</p></section><section class="card stack"><h2>② どちらで あそぶ？</h2><p>🏠 しゅくだい：1にち 1かい（20もん）</p><button type="button" class="primary full" onclick="begin('homework')" ${u.lastHomework === today() || saving ? "disabled" : ""}>${u.lastHomework === today() ?     if (screen === "play" || screen === "review") {

      KANJI = [...new Set(QUESTIONS.map(q => q.k))]; data = store.read(); render();
    } catch { app.innerHTML = `<section class="card"><h1>じゅんびを たしかめよう</h1><p>もんだい または きろくを よみこめませんでした。おとなの方に しらせてね。きろくは けしていません。</p><button type="button" onclick="location.reload()">もういちど よみこむ</button></section>`; }

  previewListen("storage", event => { if (event.key !== store.key && event.key !== null) return; try { const next = store.read(); if (next.active !== data.active) { cancelTimer(); game = review = feedback = null; screen = "home"; } data = next; render(); } catch { warning = "きろくの へんこうを よみこめません。"; render(); } });
  initialize();
})();

function initRewards(setTimeout,previewListen){
"use strict";

(() => {
  const cards = window.REWARD_CARDS;
  const tiers = [
    { id: "normal", name: "ノーマル", required: 1 },
    { id: "gold", name: "ゴールド", required: 2 },
    { id: "rainbow", name: "レインボー", required: 3 },
  ];
  const integrated = !new URLSearchParams(location.search).has("demo");
  const storageKey = integrated ? window.GameStore.key : "kanji-reading-game:rewards-demo:v1";
  let profileId = "";
  if (integrated) {
    try { profileId = window.GameStore.read().active; } catch { /* Handled by refreshState below. */ }
  }
  const storageMessage = document.getElementById("storage-message");
  const page = document.body.dataset.page;
  let state;
  let storageOK = true;
  let busy = false;
  let filter = "all";
  let lastFocus;

  function initialState() {
    return { version: 1, stars: integrated ? 0 : 3, counts: {} };
  }

  function readState() {
    if (integrated) {
      const u = window.GameStore.profile(window.GameStore.read(), profileId);
      return { version: 1, stars: u.stars, counts: { ...u.cardCounts } };
    }
    const raw = localStorage.getItem(storageKey);
    if (raw === null) return initialState();
    const value = JSON.parse(raw);
    if (!value || value.version !== 1 || !Number.isInteger(value.stars) || value.stars < 0 || value.stars > 999 || !value.counts || typeof value.counts !== "object" || Array.isArray(value.counts)) {
      throw new Error("Invalid reward state");
    }
    for (const [id, count] of Object.entries(value.counts)) {
      if (!cards.some((card) => card.id === id) || !Number.isInteger(count) || count < 0 || count > 3) throw new Error("Invalid card count");
    }
    return { version: 1, stars: value.stars, counts: { ...value.counts } };
  }

  function refreshState() {
    try {
      state = readState();
      storageOK = true;
      storageMessage.textContent = "";
    } catch {
      storageOK = false;
      state = initialState();
      storageMessage.textContent = integrated ? "とうろくや きろくを よみこめません。かんじの がめんで くみ・ばんごうを たしかめてね。" : "おためしの記録を よみこめません。ガチャは おやすみ中です。おとなの方に しらせてね。";
    }
  }

  async function saveState(next) {
    try {
      if (integrated) {
        await window.GameStore.transact(data => {
          const u = window.GameStore.profile(data, profileId);
          if (u.stars !== state.stars || JSON.stringify(u.cardCounts) !== JSON.stringify(state.counts)) throw new Error("記録が更新されました");
          u.stars = next.stars;
          u.cardCounts = { ...next.counts };
        });
      } else localStorage.setItem(storageKey, JSON.stringify(next));
      state = next;
      storageMessage.textContent = "";
      return true;
    } catch {
      storageOK = false;
      storageMessage.textContent = "ほぞんできなかったので、星もカードも かえていないよ。おとなの方に しらせてね。";
      return false;
    }
  }

  function countFor(card) {
    return state.counts[card.id] || 0;
  }

  function unlockedCount() {
    return cards.reduce((sum, card) => sum + countFor(card), 0);
  }

  function makeArt(card, tier, lazy = false) {
    const frame = document.createElement("div");
    frame.className = `card-art ${tier.id}`;
    const image = document.createElement("img");
    image.src = card.image;
    image.alt = `${card.name}の${tier.name}カード`;
    image.loading = lazy ? "lazy" : "eager";
    image.decoding = "async";
    image.addEventListener("error", () => {
      image.hidden = true;
      const message = document.createElement("p");
      message.textContent = "えを よみこめませんでした。";
      frame.append(message);
    }, { once: true });
    frame.append(image);
    if (tier.id !== "normal") {
      const sparkles = document.createElement("span");
      sparkles.className = "foil-sparkles";
      sparkles.setAttribute("aria-hidden", "true");
      for (let index = 0; index < 8; index += 1) {
        const star = document.createElement("i");
        star.textContent = "✦";
        sparkles.append(star);
      }
      frame.append(sparkles);
    }
    return frame;
  }

  function renderGacha() {
    const complete = unlockedCount() === cards.length * 3;
    document.getElementById("star-count").textContent = state.stars;
    const draw = document.getElementById("draw-card");
    draw.disabled = !storageOK || busy || complete || state.stars === 0;
    draw.textContent = complete ? "✦ ぜんぶ あつめたよ！" : "★ 1こで ハンドルをまわす";
    document.getElementById("machine-handle").disabled = draw.disabled;
    const add = document.getElementById("add-demo-stars");
    add.hidden = integrated;
    add.disabled = integrated || !storageOK || busy || complete || state.stars > 996;
    const message = document.getElementById("draw-message");
    if (complete) message.textContent = "コンプリート！ 99まいの たからものが そろったよ。";
    else if (state.stars === 0) message.textContent = integrated ? "しゅくだいや れんしゅうで 星を あつめよう！" : "おためしの星を ふやすと、また ひけるよ。";
    else if (!busy) message.textContent = "星1こで 1かい。3まい あつめた なかまは もうでないよ。";
  }

  function showResult(card, count) {
    const tier = tiers[count - 1];
    document.getElementById("result-heading").textContent = count === 1 ? "なかまが ふえたよ！" : "カードが かがやいたよ！";
    const result = document.getElementById("draw-result");
    result.replaceChildren(makeArt(card, tier));
    const title = document.createElement("h3");
    title.textContent = card.name;
    const meta = document.createElement("p");
    meta.className = "result-meta";
    meta.textContent = `${count}まいめ · ${tier.name}カード`;
    result.append(title, meta);
    document.getElementById("draw-message").textContent = `${card.name}の ${tier.name}カードを ずかんに ほぞんしたよ！`;
  }

  async function drawCard() {
    if (busy) return;
    refreshState();
    const eligible = cards.filter((card) => countFor(card) < 3);
    if (!storageOK || state.stars === 0 || eligible.length === 0) {
      renderGacha();
      return;
    }
    // おためし画面のみの等確率抽選。本番の排出率を確定するものではない。
    const random = new Uint32Array(1);
    crypto.getRandomValues(random);
    const card = eligible[Math.floor((random[0] / 4294967296) * eligible.length)];
    const count = countFor(card) + 1;
    const next = { ...state, stars: state.stars - 1, counts: { ...state.counts, [card.id]: count } };
    busy = true;
    renderGacha();
    if (!await saveState(next)) {
      busy = false;
      renderGacha();
      return;
    }
    const panel = document.querySelector(".summon-panel");
    panel.classList.add("is-drawing");
    panel.dataset.stage = "turn";
    const result = document.getElementById("draw-result");
    result.classList.remove("is-revealing");
    result.textContent = "カプセルの なかは なにかな…？";
    document.getElementById("result-heading").textContent = "ガチャガチャ…！";
    document.getElementById("draw-message").textContent = "ハンドルを ぐるっと まわしているよ！";
    const reveal = () => {
      if (!panel.isConnected) return;
      showResult(card, count);
      result.classList.add("is-revealing");
      panel.dataset.stage = "revealed";
      panel.classList.remove("is-drawing");
      busy = false;
      renderGacha();
    };
    if (matchMedia("(prefers-reduced-motion: reduce)").matches) {
      setTimeout(reveal, 80);
      return;
    }
    setTimeout(() => {
      if (!panel.isConnected) return;
      panel.dataset.stage = "drop";
      document.getElementById("draw-message").textContent = "ころん！ カプセルが でてきたよ。";
    }, 850);
    setTimeout(() => {
      if (!panel.isConnected) return;
      panel.dataset.stage = "open";
      document.getElementById("draw-message").textContent = "ぱかっ！ カプセルが ひらくよ。";
    }, 1550);
    setTimeout(reveal, 2300);
  }

  function openDetail(card, tier, button) {
    lastFocus = button;
    document.getElementById("detail-card").replaceChildren(makeArt(card, tier));
    document.getElementById("detail-name").textContent = card.name;
    document.getElementById("detail-tier").textContent = `No.${card.id} · ${tier.name}`;
    document.getElementById("detail-count").textContent = `この なかまは ${countFor(card)}まい もっているよ。`;
    document.getElementById("card-dialog").showModal();
  }

  function renderAlbum() {
    const total = unlockedCount();
    document.getElementById("collection-count").textContent = total;
    document.getElementById("collection-progress").value = total;
    const grid = document.getElementById("album-grid");
    const ownedOnly = document.getElementById("owned-only").checked;
    grid.replaceChildren();
    let visible = 0;
    let owned = 0;
    for (const card of cards) {
      for (const tier of tiers) {
        if (filter !== "all" && filter !== tier.id) continue;
        const unlocked = countFor(card) >= tier.required;
        if (ownedOnly && !unlocked) continue;
        visible += 1;
        if (unlocked) owned += 1;
        const entry = document.createElement("article");
        entry.className = "album-entry";
        entry.dataset.cardId = card.id;
        entry.dataset.tier = tier.id;
        const button = document.createElement("button");
        button.type = "button";
        button.className = "album-card-button";
        if (unlocked) {
          button.setAttribute("aria-label", `${card.name} ${tier.name}カードを おおきくみる`);
          button.append(makeArt(card, tier, true));
          button.addEventListener("click", () => openDetail(card, tier, button));
        } else {
          button.disabled = true;
          button.setAttribute("aria-label", `No.${card.id} ${tier.name} まだもっていないカード`);
          const back = document.createElement("div");
          back.className = "card-art locked-art";
          const mark = document.createElement("span");
          mark.textContent = "✧";
          mark.setAttribute("aria-hidden", "true");
          const caption = document.createElement("small");
          caption.textContent = "まだ であっていないよ";
          back.append(mark, caption);
          button.append(back);
        }
        const label = document.createElement("p");
        label.className = "album-label";
        const number = document.createElement("span");
        number.textContent = `No.${card.id}`;
        const badge = document.createElement("span");
        badge.className = `tier-label ${tier.id}`;
        badge.textContent = tier.name;
        label.append(number, badge);
        const name = document.createElement("h2");
        name.textContent = unlocked ? card.name : "？？？";
        entry.append(button, label, name);
        grid.append(entry);
      }
    }
    document.getElementById("album-empty").hidden = visible !== 0;
    document.getElementById("album-summary").textContent = `${visible}まいを ひょうじ · ${owned}まい もっているよ`;
  }

  function render() {
    if (page === "gacha") renderGacha();
    else renderAlbum();
  }

  const note = document.getElementById("profile-note");
  if (note) note.textContent = integrated ? (profileId ? `${profileId.replace("-", "くみ ")}ばんの 星とカード · この端末に保存` : "はじめに、かんじの がめんで くみ・ばんごうを とうろくしてね。") : "おためし · がくしゅうの星とは べつの記録です。";
  for (const element of document.querySelectorAll("[data-demo-only]")) element.hidden = integrated;
  if (!integrated) for (const link of document.querySelectorAll('a[href="gacha.html"], a[href="collection.html"]')) link.href += "?demo=1";
  refreshState();
  render();
  if (page === "gacha") {
    document.getElementById("draw-card").addEventListener("click", drawCard);
    document.getElementById("machine-handle").addEventListener("click", drawCard);
    document.getElementById("add-demo-stars").addEventListener("click", async () => {
      if (integrated || busy) return;
      refreshState();
      if (storageOK && state.stars <= 996 && unlockedCount() < 99) await saveState({ ...state, stars: state.stars + 3 });
      renderGacha();
    });
  } else {
    document.getElementById("tier-filters").addEventListener("click", (event) => {
      const button = event.target.closest("button[data-filter]");
      if (!button) return;
      filter = button.dataset.filter;
      for (const item of document.querySelectorAll("[data-filter]")) item.setAttribute("aria-pressed", String(item === button));
      renderAlbum();
    });
    document.getElementById("owned-only").addEventListener("change", renderAlbum);
    const dialog = document.getElementById("card-dialog");
    document.getElementById("close-detail").addEventListener("click", () => dialog.close());
    document.getElementById("detail-done").addEventListener("click", () => dialog.close());
    dialog.addEventListener("close", () => lastFocus?.isConnected && lastFocus.focus());
  }
  document.getElementById("reset-demo").addEventListener("click", () => {
    if (integrated || busy || !confirm("おためしの星とカードを すべてリセットしますか？ 学習の記録は変更しません。")) return;
    try {
      localStorage.removeItem(storageKey);
      refreshState();
      if (page === "gacha") {
        document.getElementById("draw-result").textContent = "ガチャをひくと、ここにカードが あらわれるよ。";
        document.getElementById("result-heading").textContent = "きみの なかまを よぼう";
      } else document.getElementById("card-dialog").close();
      render();
    } catch {
      storageMessage.textContent = "リセットできませんでした。";
    }
  });
  previewListen("storage", (event) => {
    if (event.key !== storageKey && event.key !== null) return;
    refreshState();
    if (page === "collection") document.getElementById("card-dialog").close();
    render();
  });
})();

let timers = [], listeners;
function showScreen() {
  for (const timer of timers) clearTimeout(timer);
  timers = [];
  listeners?.abort(); listeners = new AbortController();
  const requested = location.hash.slice(1).split('?')[0];
  const screen = Object.hasOwn(screens, requested) ? requested : 'game';
  document.body.dataset.page = screen;
  document.getElementById('screen-style').textContent = styles[screen];
  document.body.innerHTML = screens[screen];
  const schedule = (fn, ms) => {const timer = window.setTimeout(fn, ms); timers.push(timer); return timer;};
  const listen = (event, fn) => window.addEventListener(event, fn, {signal: listeners.signal});
  if (screen === 'game') initApp(async () => new Response(questionCSV, {status: 200}), schedule, listen);
  else initRewards(schedule, listen);
  scrollTo(0, 0);
}
window.addEventListener('hashchange', () => {if (Object.hasOwn(screens, location.hash.slice(1).split('?')[0])) showScreen();});
